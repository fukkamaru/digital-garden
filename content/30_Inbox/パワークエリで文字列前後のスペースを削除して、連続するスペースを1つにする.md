(old_name) =>

let
    //文字列の前後のスペースを削除、半角スペースを全角スペースに置換する
    trimmed_name = Text.Trim(old_name),
    harf_space_replace_to_full = Text.Replace(trimmed_name, " ", "　"),

    //正規表現で連続する全角スペースを1つの全角スペースに置換
    consolidated_spaces = Text.Combine(List.Transform(Text.SplitAny(harf_space_replace_to_full, "　"), each if _ = "" then null else _), "　")
in
    consolidated_spaces
